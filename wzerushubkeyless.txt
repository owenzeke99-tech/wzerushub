-- Hàm kiểm tra xem đã lưu key chưa và còn hạn 24h không
local function hasValidKey()
    if not (writefile and readfile and isfile) then return false end
    if isfile(SAVE_FILE) then
        local success, data = pcall(function()
            return game:GetService("HttpService"):JSONDecode(readfile(SAVE_FILE))
        end)
        if success and data and data.ExpiryTime then
            if os.time() < data.ExpiryTime then
                return true
            else
                -- Hết hạn 24h thì xóa file cũ
                pcall(function() delfile(SAVE_FILE) end)
            end
        end
    end
    return false
end

-- Hàm lưu key với thời hạn 24 giờ (86400 giây)
local function saveKeyData()
    if writefile and game:GetService("HttpService") then
        pcall(function()
            local data = {
                ExpiryTime = os.time() + 86400 -- 24 giờ tới
            }
            writefile(SAVE_FILE, game:GetService("HttpService"):JSONEncode(data))
        end)
    end
end

-- Nếu đã lưu key hợp lệ thì load thẳng script chính luôn, không hiện bảng key nữa
